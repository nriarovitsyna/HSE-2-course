# Яровицына Наталья БПИ244
.data
codes: 
    # Коды для 7-сегментного индикатора: 0-9, A-F
    .byte 0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07, 0x7F, 0x6F, 0x77, 0x7C, 0x39, 0x5E, 0x79, 0x71

.text
.globl main

main:
    li s0, 0                # Счетчик текущей цифры (0-15)
    li s1, 0                # Флаг индикатора: 0 = правый, 1 = левый
    li s2, 16               # Максимальное значение цифры + 1 (для сброса)
    
loop:
    # Подготовка аргументов для вызова подпрограммы отображения
    mv a0, s0               # Передаем текущую цифру в a0
    mv a1, s1               # Передаем выбор индикатора в a1
    
    # Вызов подпрограммы отображения цифры
    jal display
    
    # Задержка 1000 мс (1 секунда)
    li a0, 1000             # Время задержки в миллисекундах
    li a7, 32               # Номер системного вызова для sleep
    ecall                   # Вызов системы для задержки
    
    # Переключаем индикатор для следующей итерации
    xori s1, s1, 1          # Инвертируем бит: 0->1, 1->0
    
    # Увеличиваем счетчик цифр только после отображения на обоих индикаторах
    bnez s1, continue_loop  # Если индикатор = 1 (левый), пропускаем увеличение
    addi s0, s0, 1          # Увеличиваем цифру (только после правого индикатора)
    
    # Проверяем, не достигли ли максимального значения
    blt s0, s2, continue_loop # Если цифра < 16, продолжаем
    li s0, 0                # Сбрасываем счетчик на 0 при достижении 16
    
continue_loop:
    j loop                  # Бесконечный цикл

# Подпрограмма отображения цифры на 7-сегментном индикаторе
# Входные параметры:
#   a0 - цифра для отображения (0-15)
#   a1 - выбор индикатора: 0 = правый, 1 = левый
display:
    # Сохраняем используемые регистры в стеке
    addi sp, sp, -12
    sw t0, 0(sp)
    sw t1, 4(sp)
    sw t2, 8(sp)
    
    # Получаем код сегментов для текущей цифры
    la t0, codes            # Загружаем адрес таблицы кодов
    andi t1, a0, 0x0F       # Берем только младшие 4 бита (обеспечиваем диапазон 0-15)
    add t2, t0, t1          # Вычисляем адрес нужного кода в таблице
    lb t0, 0(t2)            # Загружаем код сегментов для цифры
    
    # Проверяем, нужно ли добавить точку (для цифр A-F)
    li t1, 9                # Пороговое значение для точки
    ble a0, t1, no_dot      # Если цифра <= 9, точку не добавляем
    ori t0, t0, 0x80        # Добавляем точку (8-й бит) для цифр A-F (10-15)
    
no_dot:
    # Выбираем индикатор для вывода на основе параметра a1
    beqz a1, right_display  # Если a1 = 0, выводим на правый индикатор
    
    # Вывод на левый индикатор
    li t1, 0xFFFF0011       # Адрес левого индикатора
    sb t0, 0(t1)            # Записываем код сегментов
    j display_done
    
right_display:
    # Вывод на правый индикатор
    li t1, 0xFFFF0010       # Адрес правого индикатора
    sb t0, 0(t1)            # Записываем код сегментов
    
display_done:
    # Восстанавливаем регистры из стека
    lw t0, 0(sp)
    lw t1, 4(sp)
    lw t2, 8(sp)
    addi sp, sp, 12
    ret                     # Возврат из подпрограммы
