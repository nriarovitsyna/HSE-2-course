.text

# Функция вычисления квадратного корня методом Ньютона
# Входные параметры:
#   fa0 - число для извлечения корня (положительное)
#   fa1 - требуемая точность вычисления
# Выходное значение:
#   fa0 - вычисленный квадратный корень
calculate_sqrt:
    # Пролог функции - сохранение регистров в стеке
    addi sp, sp, -20      # Резервируем 20 байт в стеке (5 слов)
    fsw fs0, 16(sp)       # Сохраняем сохраненный регистр fs0
    fsw fs1, 12(sp)       # Сохраняем сохраненный регистр fs1  
    fsw fs2, 8(sp)        # Сохраняем сохраненный регистр fs2
    sw ra, 4(sp)          # Сохраняем адрес возврата
    sw s0, 0(sp)          # Сохраняем сохраненный регистр s0
    
    # Инициализация рабочих регистров
    fmv.s fs0, fa0        # fs0 = исходное число (сохраняем параметр)
    fmv.s fs1, fa1        # fs1 = требуемая точность (сохраняем параметр)
    
    # Подготовка констант для вычислений
    fcvt.s.w ft4, zero    # ft4 = 0.0 (константа ноль)
    li t0, 1              
    fcvt.s.w ft5, t0      # ft5 = 1.0 (константа единица)
    li t0, 2
    fcvt.s.w ft6, t0      # ft6 = 2.0 (константа два)
    
    # Проверка особого случая: число = 0
    feq.s t0, fs0, ft4    # Проверяем, равно ли число нулю
    bnez t0, zero_case    # Если да, переходим к обработке нуля
    
    # Начальное приближение для метода Ньютона: x0 = число / 2
    fmv.s fs2, fs0        # fs2 = текущее приближение (x_n)
    fdiv.s fs2, fs2, ft6  # fs2 = число / 2.0
    
    li s0, 0              # s0 = счетчик итераций (инициализируем 0)
            
# Основной цикл итеративного уточнения корня
loop:
    # Итерация метода Ньютона: x_{n+1} = (x_n + число/x_n) / 2
    fdiv.s ft1, fs0, fs2  # ft1 = число / x_n
    fadd.s ft1, fs2, ft1  # ft1 = x_n + (число / x_n)
    fdiv.s ft1, ft1, ft6  # ft1 = (x_n + число/x_n) / 2 (новое приближение)
    
    # Проверка достижения требуемой точности
    fsub.s ft2, ft1, fs2  # ft2 = разность между новым и старым приближением
    fabs.s ft2, ft2       # ft2 = |x_{n+1} - x_n| (абсолютное значение)
    
    # Сравниваем достигнутую точность с требуемой
    flt.s t0, ft2, fs1     # Проверяем: |x_{n+1} - x_n| < точность?
    bnez t0, return_result # Если точность достигнута, выходим из цикла
    
    # Проверка максимального количества итераций (защита от бесконечного цикла)
    addi s0, s0, 1        # Увеличиваем счетчик итераций
    li t1, 100            # Максимальное число итераций = 100
    bge s0, t1, return_result # Если превысили лимит, выходим
    
    # Подготовка к следующей итерации
    fmv.s fs2, ft1        # Обновляем текущее приближение: x_n = x_{n+1}
    j loop                # Переходим к следующей итерации

# Обработка особого случая: квадратный корень из нуля
zero_case:
    fmv.s fs2, ft4        # Корень из 0 = 0

# Завершение функции - возврат результата
return_result:
    fmv.s fa0, fs2        # Возвращаем результат в fa0
    
    # Эпилог функции - восстановление регистров из стека
    lw s0, 0(sp)          # Восстанавливаем s0
    lw ra, 4(sp)          # Восстанавливаем адрес возврата
    flw fs2, 8(sp)        # Восстанавливаем fs2
    flw fs1, 12(sp)       # Восстанавливаем fs1
    flw fs0, 16(sp)       # Восстанавливаем fs0
    addi sp, sp, 20       # Восстанавливаем указатель стека
    ret                   # Возврат из функции
