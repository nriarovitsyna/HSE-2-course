# macros.asm - Макроопределения для программы вычисления квадратного корня

# Макрос для чтения вещественного числа с плавающей точкой
# %msg - метка строки-приглашения для ввода
# %reg - регистр для сохранения введенного значения
.macro read_float(%msg, %reg)
    la a0, %msg           # Загружаем адрес строки-приглашения
    li a7, 4              # Код системного вывода для печати строки
    ecall                 # Выводим приглашение
    li a7, 6              # Код системного ввода для чтения float
    ecall                 # Читаем число (результат в fa0)
    fmv.s %reg, fa0       # Сохраняем введенное значение в указанный регистр
.end_macro

# Макрос для проверки положительности чисел
# %num_reg - регистр с проверяемым числом
# %prec_reg - регистр с проверяемой точностью
# %fail - метка для перехода при отрицательном значении
.macro check_positive(%num_reg, %prec_reg, %fail)
    fmv.s ft0, %num_reg   # Копируем число во временный регистр
    fcvt.s.w ft1, zero    # Конвертируем 0 в float (создаем эталон 0.0)
    flt.s t1, ft0, ft1    # Сравниваем: число < 0?
    bnez t1, %fail        # Если да, переходим к обработке ошибки
    
    fmv.s ft0, %prec_reg  # Копируем точность во временный регистр
    flt.s t1, ft0, ft1    # Сравниваем: точность < 0?
    bnez t1, %fail        # Если да, переходим к обработке ошибки
.end_macro

# Макрос для вывода результата вычислений
# %res_reg - регистр с результатом для вывода
# %msg - метка строки-сообщения перед результатом
.macro print_result(%res_reg, %msg)
    addi sp, sp, -4       # Резервируем место в стеке
    fsw %res_reg, 0(sp)   # Сохраняем результат в стек
    
    la a0, %msg           # Загружаем адрес строки-сообщения
    li a7, 4              # Код системного вывода для печати строки
    ecall                 # Выводим сообщение
    
    flw fa0, 0(sp)        # Восстанавливаем результат из стека в fa0
    addi sp, sp, 4        # Освобождаем стек
    li a7, 2              # Код системного вывода для печати float
    ecall                 # Выводим результат
    
    la a0, new_line       # Загружаем адрес символа новой строки
    li a7, 4              # Код системного вывода для печати строки
    ecall                 # Выводим перевод строки
.end_macro

# Макрос для вычисления квадратного корня
# %num_reg - регистр с числом для извлечения корня
# %prec_reg - регистр с требуемой точностью
# %res_reg - регистр для сохранения результата
.macro compute_sqrt(%num_reg, %prec_reg, %res_reg)
    fmv.s fa0, %num_reg   # Передаем число как первый аргумент
    fmv.s fa1, %prec_reg  # Передаем точность как второй аргумент
    jal calculate_sqrt    # Вызываем функцию вычисления корня
.end_macro

.data
# Строковые константы программы
enter_num_msg: .asciz "Введите число (положительное): "
enter_prec_msg: .asciz "Введите погрешность вычислений: "
result_msg: .asciz "Квадратный корень = "
error_msg: .asciz "ОШИБКА: Введены некорректные данные! Число и точность должны быть > 0!\n"
new_line: .asciz "\n"
