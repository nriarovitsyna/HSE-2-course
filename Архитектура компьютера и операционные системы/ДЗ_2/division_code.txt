# Яровицына Наталья
.data
input_dividend:  .asciz "Введите делимое: "             # Сообщение для ввода делимого
input_divisor:   .asciz "Введите делитель: "            # Сообщение для ввода делителя
error_msg:       .asciz "ОШИБКА: Деление на ноль!\n"    # Сообщение об ошибке
quotient:        .asciz "Частное = "                    # Текст для вывода частного
remainder:       .asciz "Остаток = "                    # Текст для вывода остатка
nextline:        .asciz "\n"                            # Перевод строки

.text
    # Вывод приглашения для ввода делимого
    li a7, 4                   # Системный вызов для печати строки
    la a0, input_dividend      # Загрузка адреса строки приглашения
    ecall                      # Выполнение системного вызова

    # Чтение делимого
    li a7, 5                   # Системный вызов для чтения целого числа
    ecall                      # Чтение числа, результат записывается в a0
    mv s0, a0                  # Сохранение делимого в s0

    # Вывод приглашения для ввода делителя
    li a7, 4                   # Системный вызов для печати строки
    la a0, input_divisor       # Загрузка адреса строки приглашения
    ecall                      # Выполнение системного вызова

    # Чтение делителя
    li a7, 5                   # Системный вызов для чтения целого числа
    ecall                      # Чтение числа, результат записывается в a0
    mv s1, a0                  # Сохранение делителя в s1

    # Проверка деления на ноль
    beqz s1, div_by_zero       

    # Инициализация переменных
    li s2, 0                   # Инициализация частного нулем
    
    # Определение знака частного
    slt t0, s0, zero           # t0 = 1 если делимое отрицательное
    slt t1, s1, zero           # t1 = 1 если делитель отрицательное
    xor s3, t0, t1             # s3 = 1 если знаки разные

    # Вычисление модуля делимого
    mv t2, s0                  # Копирование делимого в t2
    bgez t2, div_abs           # Если число положительное, переход
    neg t2, t2                 # Преобразование в положительное число
    
div_abs:
    # Вычисление модуля делителя
    mv t3, s1                  # Копирование делителя в t3
    bgez t3, division          # Если число положительное, переход
    neg t3, t3                 # Преобразование в положительное число

division:
    # Цикл деления через вычитание
    blt t2, t3, fix_signs      # Выход если делимое меньше делителя
    sub t2, t2, t3             # Вычитание делителя из делимого
    addi s2, s2, 1             # Увеличение частного на 1
    j division                 # Повтор цикла

fix_signs:
    # Исправление знака частного
    beqz s3, check_remainder   # Если знаки одинаковые, пропускаем
    neg s2, s2                 # Меняем знак частного

check_remainder:
    # Проверка знака остатка
    bgez s0, output_res        # Если делимое положительное, пропускаем  
    neg t2, t2                 # Меняем знак остатка
    
output_res:
    # Вывод значения частного
    li a7, 4                   # Системный вызов для печати строки
    la a0, quotient            # Загрузка адреса строки quotient
    ecall                      # Вывод строки
    li a7, 1                   # Системный вызов для печати целого числа
    mv a0, s2                  # Загрузка значения частного
    ecall                      # Вывод числа

    # Вывод перевода строки
    li a7, 4                   # Системный вызов для печати строки
    la a0, nextline            # Загрузка адреса перевода строки
    ecall                      # Вывод перевода строки
    
    # Вывод значения остатка
    li a7, 4                   # Системный вызов для печати строки
    la a0, remainder           # Загрузка адреса строки remainder
    ecall                      # Вывод строки
    li a7, 1                   # Системный вызов для печати целого числа
    mv a0, t2                  # Загрузка значения остатка
    ecall                      # Вывод числа

    # Вывод перевода строки
    li a7, 4                   # Системный вызов для печати строки
    la a0, nextline            # Загрузка адреса перевода строки
    ecall                      # Вывод перевода строки

    j finish                   # Переход к метке finish

div_by_zero:
    # Обработка ошибки деления на ноль
    li a7, 4                   # Системный вызов для печати строки
    la a0, error_msg           # Загрузка адреса строки error_msg
    ecall                      # Вывод строки

finish:
    # Завершение программы
    li a7, 10                  # Системный вызов для выхода
    ecall                      # Завершение программы
