# Example that calculates 
# the Fibonacci sequence.
main:
    mv t0, zero           # Псевдоинструкция: устанавливает значение t0 равным 0
    li t1, 1              # Псевдоинструкция: загружает в t1 число 1
    
    li a7, 5              # Псевдоинструкция: загружает в a7 число 5
    ecall                 # Системный вызов: выполняет чтение целого числа, вводимое пользователем, результат сохраняется в a0
    mv t3, a0             # Псевдоинструкция: копирует значение из a0 в t3
fib:
    beqz t3, finish       # Псевдоинструкция: если t3 равен 0, перейти к метке finish
    add t2, t1, t0        # Инструкция: устанавливает значение t2 равным t1 + t0
    mv t0, t1             # Псевдоинструкция: копирует значение из t1 в t0
    mv t1, t2             # Псевдоинструкция: копирует значение из t2 в t1
    addi t3, t3, -1       # Инструкция: уменьшает значение t3 на 1
    j fib                 # Псевдоинструкция: переход к метке fib
finish:
    li a7, 1              # Псевдоинструкция: загружает в a7 число 1
    mv a0, t0             # Псевдоинструкция: копирует значение из t0 в a0
    ecall                 # Системный вызов: выводит целое число из a0 на экран
